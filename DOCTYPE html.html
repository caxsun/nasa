<script>
    // 사용자 개인 API Key 설정
    const API_KEY = 'mF966tNZ5cUDXx99W1vypGQJcct0a2FCNkDC8B9k';
    // 기본 날짜 설정 (오늘 날짜)
    window.onload = function() {
        const today = new Date().toISOString().split('T')[0];
        document.getElementById('startDate').value = today;
        document.getElementById('endDate').value = today;
        fetchAsteroids();
    };
    async function fetchAsteroids() {
        const startDate = document.getElementById('startDate').value;
        const endDate = document.getElementById('endDate').value;
        const summaryDiv = document.getElementById('summary');
        const resultsDiv = document.getElementById('results');
        if (!startDate) {
            alert('시작 날짜를 선택해주세요.');
            return;
        }
        // 초기화 및 로딩 표시
        summaryDiv.innerHTML = '';
        resultsDiv.innerHTML = '<div class="loading">NASA API에서 데이터를 불러오는 
중...</div>';
        const url = 
`https://api.nasa.gov/neo/rest/v1/feed?start_date=${startDate}&end_date=${endDate}&
api_key=${API_KEY}`;
        try {
            const response = await fetch(url);
            
            if (!response.ok) {
                throw new Error(`요청 실패 (상태 코드: ${response.status})`);
            }
            const data = await response.json();
            displayResults(data);
        } catch (error) {
            resultsDiv.innerHTML = `<div class="error">데이터를 가져오는 중 오류가 발
생했습니다: ${error.message}</div>`;
        }
    }
    function displayResults(data) {
        const summaryDiv = document.getElementById('summary');
        const resultsDiv = document.getElementById('results');
        
        summaryDiv.innerHTML = `총 <strong>${data.element_count}</strong>개의 소
행성이 발견되었습니다.`;
        resultsDiv.innerHTML = '';
        const nearEarthObjects = data.near_earth_objects;
        // 날짜별로 정렬되어 나오는 소행성 목록 순회
        Object.keys(nearEarthObjects).forEach(date => {
            nearEarthObjects[date].forEach(asteroid => {
                const isHazardous = asteroid.is_potentially_hazardous_asteroid;
                
                // 추정 지름 (m 단위)
                const minDiameter = 
Math.round(asteroid.estimated_diameter.meters.estimated_diameter_min);
                const maxDiameter = 
Math.round(asteroid.estimated_diameter.meters.estimated_diameter_max);
                
                // 가장 가까운 접근 정보
                const closeApproach = asteroid.close_approach_data[0];
                const velocityKmH = 
Math.round(closeApproach.relative_velocity.kilometers_per_hour).toLocaleString();
                const missDistanceKm = 
Math.round(closeApproach.miss_distance.kilometers).toLocaleString();
                // 카드 HTML 생성
                const cardHtml = `
                    <div class="card ${isHazardous ? 'hazardous' : ''}">
                        <div>
                            <span class="badge ${isHazardous ? 'badge-danger' : 
'badge-safe'}">
                                ${isHazardous ? '⚠ 위협적 소행성' : '✅ 안전함'}
                            </span>
                            <h3>${asteroid.name}</h3>
                            <div class="info-item"><span>접근 날짜:</span> 
${closeApproach.close_approach_date_full}</div>
                            <div class="info-item"><span>추정 크기:</span> 
${minDiameter}m ~ ${maxDiameter}m</div>
                            <div class="info-item"><span>상대 속도:</span> 
${velocityKmH} km/h</div>
                            <div class="info-item"><span>최대 접근 거리:</span> 
${missDistanceKm} km</div>
                            <div class="info-item"><span>절대 등급:</span> 
${asteroid.absolute_magnitude_h} H</div>
                        </div>
                        <a href="${asteroid.nasa_jpl_url}" target="_blank" 
class="link-btn">JPL 상세 정보 보기 🔗</a>
                    </div>
                `;
                resultsDiv.innerHTML += cardHtml;
            });
        });
    }
